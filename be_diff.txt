+import { PrismaClient } from '@prisma/client'
+import bcrypt from 'bcryptjs'
+import * as xlsx from 'xlsx'
+import crypto from 'crypto'
+import { IEmailService } from '../../../../shared/application/email.service.interface.js'
+import { AppError } from '../../../../shared/application/app.error.js'
+
+const prisma = new PrismaClient()
+
+interface ImportStudentRow {
+    MSSV?: string;
+    'Hß╗ì v├á t├¬n'?: string;
+    Email?: string;
+    'Sß╗æ ─æiß╗çn thoß║íi'?: string;
+    'Kß╗│ hß╗ìc'?: string;
+    'Lß╗¢p hß╗ìc'?: string;
+}
+
+export class ImportStudentsExcelUseCase {
+    constructor(private readonly emailService: IEmailService) {}
+
+    async execute(input: { fileBuffer: Buffer; fileName: string; fileUrl: string; importedByUserId: string }) {
+        const { fileBuffer, fileName, fileUrl, importedByUserId } = input
+        
+        let successCount = 0
+        let errorCount = 0
+        const errors: string[] = []
+
+        // 1. Create ImportBatch record first
+        const batch = await prisma.importBatch.create({
+            data: {
+                FileName: fileName,
+                FileUrl: fileUrl,
+                Status: 'PROCESSING',
+                TotalRows: 0,
+                SuccessCount: 0,
+                ErrorCount: 0,
+                ImportedBy: importedByUserId
+            }
+        })
+
+        try {
+            // 2. Parse Excel file
+            const workbook = xlsx.read(fileBuffer, { type: 'buffer' })
+            const sheetName = workbook.SheetNames[0]
+            if (!sheetName) {
+                throw new AppError('INVALID_FILE', 'File Excel kh├┤ng c├│ dß╗» liß╗çu', 400)
+            }
+
+            const sheet = workbook.Sheets[sheetName]
+            const rows: ImportStudentRow[] = xlsx.utils.sheet_to_json(sheet)
+            
+            await prisma.importBatch.update({
+                where: { Id: batch.Id },
+                data: { TotalRows: rows.length }
+            })
+
+            const studentRole = await prisma.role.findUnique({ where: { RoleName: 'student' } })
+            if (!studentRole) {
+                throw new AppError('SYSTEM_ERROR', 'Ch╞░a cß║Ñu h├¼nh role student trong hß╗ç thß╗æng', 500)
+            }
+
+            // Process row by row
+            for (let i = 0; i < rows.length; i++) {
+                const row = rows[i]
+                const rowIndex = i + 2 // +2 because 0-index and header row
+
+                try {
+                    const mssv = row['MSSV']?.toString().trim()
+                    const fullName = row['Hß╗ì v├á t├¬n']?.toString().trim()
+                    const email = row['Email']?.toString().trim()
+                    const phone = row['Sß╗æ ─æiß╗çn thoß║íi']?.toString().trim()
+                    const semesterCode = row['Kß╗│ hß╗ìc']?.toString().trim()
+                    const classCode = row['Lß╗¢p hß╗ìc']?.toString().trim()
+
+                    if (!mssv || !fullName || !email || !semesterCode || !classCode) {
+                        throw new Error('Thiß║┐u th├┤ng tin bß║»t buß╗Öc (MSSV, Hß╗ì v├á t├¬n, Email, Kß╗│ hß╗ìc, Lß╗¢p hß╗ìc)')
+                    }
+
+                    // 1. Check duplicate MSSV or Email
+                    const existingUser = await prisma.user.findFirst({
+                        where: {
+                            OR: [
+                                { Email: email },
+                                { StudentCode: mssv }
+                            ]
+                        }
+                    })
+
+                    if (existingUser) {
+                        throw new Error(`Email hoß║╖c MSSV ─æ├ú tß╗ôn tß║íi trong hß╗ç thß╗æng`)
+                    }
+
+                    // 2. Check Semester
+                    const semester = await prisma.semester.findFirst({
+                        where: { Code: semesterCode }
+                    })
+
+                    if (!semester) {
+                        throw new Error(`Kh├┤ng t├¼m thß║Ñy Kß╗│ hß╗ìc "${semesterCode}" trong hß╗ç thß╗æng`)
+                    }
+
+                    // 3. Find Classes
+                    const classes = await prisma.class.findMany({
+                        where: {
+                            SemesterId: semester.Id,
+                            ClassCode: classCode
+                        },
+                        include: {
+                            Subject: true
+                        }
+                    })
+
+                    if (classes.length === 0) {
+                        throw new Error(`Kh├┤ng t├¼m thß║Ñy Lß╗¢p hß╗ìc "${classCode}" thuß╗Öc Kß╗│ hß╗ìc "${semesterCode}"`)
+                    }
+
+                    // 4. Create User
+                    const rawPassword = crypto.randomBytes(4).toString('hex') // 8 chars
+                    const passwordHash = await bcrypt.hash(rawPassword, 10)
+
+                    // Note: Since we want LastLoginAt to be null for FirstLogin=true, it will be null by default
+                    await prisma.user.create({
+                        data: {
+                            StudentCode: mssv,
+                            FullName: fullName,
+                            Email: email,
+                            Phone: phone,
+                            PasswordHash: passwordHash,
+                            Status: 'Active',
+                            UserRole: {
+                                create: {
+                                    RoleId: studentRole.Id,
+                                    AssignedAt: new Date()
+                                }
+                            },
+                            // Add student to all matching classes (subjects)
+                            StudentClass: {
+                                create: classes.map(cls => ({
+                                    ClassId: cls.Id,
+                                    EnrolledAt: new Date()
+                                }))
+                            }
+                        }
+                    })
+
+                    // 5. Send Email
+                    const subjectsList = classes.map(c => c.Subject?.SubjectCode).filter(Boolean).join(', ')
+                    
+                    await this.emailService.sendEmail(
+                        email,
+                        'Th├┤ng tin t├ái khoß║ún sinh vi├¬n AITA',
+                        `
+                            <p>Ch├áo <strong>${fullName}</strong>,</p>
+                            <p>T├ái khoß║ún cß╗ºa bß║ín tr├¬n hß╗ç thß╗æng AITA ─æ├ú ─æ╞░ß╗úc tß║ío th├ánh c├┤ng.</p>
+                            <ul>
+                                <li><strong>MSSV:</strong> ${mssv}</li>
+                                <li><strong>Email ─æ─âng nhß║¡p:</strong> ${email}</li>
+                                <li><strong>Mß║¡t khß║⌐u tß║ím:</strong> ${rawPassword}</li>
+                                <li><strong>Kß╗│ hß╗ìc hiß╗çn tß║íi:</strong> Semester ${semesterCode}</li>
+                                <li><strong>Lß╗¢p hß╗ìc:</strong> ${classCode}</li>
+                                <li><strong>C├íc m├┤n tham gia:</strong> ${subjectsList || 'Kh├┤ng c├│'}</li>
+                            </ul>
+                            <p>Vui l├▓ng ─æ─âng nhß║¡p v├á thay ─æß╗òi mß║¡t khß║⌐u trong lß║ºn ─æ─âng nhß║¡p ─æß║ºu ti├¬n ─æß╗â ─æß║úm bß║úo an to├án th├┤ng tin.</p>
+                        `
+                    )
+
+                    successCount++
+                } catch (err: any) {
+                    errorCount++
+                    errors.push(`D├▓ng ${rowIndex} (${row.Email || 'Kh├┤ng r├╡'}): ${err.message}`)
+                }
+            }
+
+            // Update batch final status
+            const finalBatch = await prisma.importBatch.update({
+                where: { Id: batch.Id },
+                data: {
+                    Status: 'COMPLETED',
+                    SuccessCount: successCount,
+                    ErrorCount: errorCount,
+                    ErrorDetails: JSON.stringify(errors)
+                }
+            })
+
+            return {
+                batchId: finalBatch.Id,
+                totalProcessed: rows.length,
+                successCount,
+                errorCount,
+                errors
+            }
+
+        } catch (err: any) {
+            await prisma.importBatch.update({
+                where: { Id: batch.Id },
+                data: {
+                    Status: 'FAILED',
+                    ErrorDetails: err.message
+                }
+            })
+            throw err
+        }
+    }
+}
